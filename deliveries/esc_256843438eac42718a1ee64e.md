#!/usr/bin/env python3
"""
Single-file USGS Earthquake Weekly Report & Two-Route Cross-Check CLI.
Pulls earthquakes from https://earthquake.usgs.gov/fdsnws/event/1/query (format=text and
format=geojson&orderby=magnitude) and verifies weekly totals via /fdsnws/event/1/count.
Generates Rule-25 CSV outputs, structured JSON audit snapshots, and a complete Markdown report.

import argparse
import csv
import datetime
import hashlib
import json
import os
import sys
import time
import urllib.error
import urllib.parse
import urllib.request

USGS_HOST = "https://earthquake.usgs.gov"
QUERY_PATH = "/fdsnws/event/1/query"
COUNT_PATH = "/fdsnws/event/1/count"
CSV_COLUMNS = [
    "event_id",
    "time_utc",
    "magnitude",
    "magType",
    "depth_km",
    "place",
    "week_start",

def parse_iso_utc_to_epoch_ms(ts_str: str) -> int:
    s = ts_str.strip()
    if s.endswith("Z"):
        s = s[:-1] + "+00:00"
    elif "+" not in s[10:] and "-" not in s[10:]:
        s = s + "+00:00"
    dt = datetime.datetime.fromisoformat(s)
    return int(round(dt.timestamp() * 1000))


def fetch_usgs(path: str, params_list: list, timeout: int = 45) -> dict:
    qs = urllib.parse.urlencode(params_list)
    url = f"{USGS_HOST}{path}?{qs}"
    last_err = None
    for attempt in range(2):
        retrieved_utc = datetime.datetime.now(datetime.timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")
        req = urllib.request.Request(
            url,
            headers={"User-Agent": "USGS-Quake-Weekly-CLI/1.0", "Accept": "*/*"},
        )
        try:
            with urllib.request.urlopen(req, timeout=timeout) as resp:
                raw_bytes = resp.read()
                http_date = resp.headers.get("Date", "")
                http_status = resp.status
                text = raw_bytes.decode("utf-8", errors="replace")
                return {
                    "ok": True,
                    "url": url,
                    "host": "earthquake.usgs.gov",
                    "endpoint": path,
                    "params": dict(params_list),
                    "retrieved_utc": retrieved_utc,
                    "http_status": http_status,
                    "http_date": http_date,
                    "text": text,
                }
        except urllib.error.HTTPError as e:
            err_body = e.read().decode("utf-8", errors="replace")
            http_date = e.headers.get("Date", "") if e.headers else ""
            last_err = {
                "ok": False,
                "url": url,
                "host": "earthquake.usgs.gov",
                "endpoint": path,
                "params": dict(params_list),
                "retrieved_utc": retrieved_utc,
                "http_status": e.code,
                "http_date": http_date,
                "error": f"HTTP {e.code}: {err_body[:500]}",
                "text": err_body,
            }
        except Exception as e:
            last_err = {
                "ok": False,
                "url": url,
                "host": "earthquake.usgs.gov",
                "endpoint": path,
                "params": dict(params_list),
                "retrieved_utc": retrieved_utc,
                "http_status": 0,
                "http_date": "",
                "error": str(e),
                "text": str(e),
            }
        if attempt == 0:
            time.sleep(2)
    return last_err


def parse_text_events(raw_text: str, week_start_date: str, end_exclusive_iso: str) -> list:
    lines = [ln for ln in raw_text.splitlines() if ln.strip()]
    if not lines:
        return []
    rows = []
    for ln in lines[1:]:
        if ln.startswith("#"):
            continue
        parts = ln.split("|", 12)
        if len(parts) < 13:
            continue
        event_id = parts[0].strip()
        time_utc = parts[1].strip()
        # Guard against inclusive boundary edge: drop if stamped >= next Monday 00:00:00 UTC
        if time_utc >= end_exclusive_iso:
            continue
        depth_raw = parts[4].strip()
        mag_type = parts[9].strip()
        mag_raw = parts[10].strip()
        place = parts[12].strip()
        rows.append(
            {
                "event_id": event_id,
                "time_utc": time_utc,
                "magnitude": mag_raw,
                "magType": mag_type,
                "depth_km": depth_raw,
                "place": place,
                "week_start": week_start_date,
            }
        )
    return rows


def parse_geojson_events(raw_json_text: str, end_exclusive_ms: int) -> tuple:
    obj = json.loads(raw_json_text)
    meta_count = (obj.get("metadata") or {}).get("count")
    generated_ms = (obj.get("metadata") or {}).get("generated")
    events = []
    for idx, feat in enumerate(obj.get("features") or []):
        props = feat.get("properties") or {}
        geom = feat.get("geometry") or {}
        coords = geom.get("coordinates") or [None, None, None]
        t_ms = props.get("time")
        if t_ms is not None and int(t_ms) >= end_exclusive_ms:
            continue
        mag = props.get("mag")
        depth = coords[2] if len(coords) > 2 else None
        events.append(
            {
                "raw_rank_index": idx,
                "event_id": str(feat.get("id") or ""),
                "time_ms": int(t_ms) if t_ms is not None else 0,
                "mag": float(mag) if mag is not None else None,
                "mag_raw": "" if mag is None else str(mag),
                "depth": float(depth) if depth is not None else None,
                "depth_raw": "" if depth is None else str(depth),
                "place": "" if props.get("place") is None else str(props.get("place")),
            }
        )
    return meta_count, generated_ms, events


def compute_bands(events: list) -> dict:
    bands = {
        "below_2_0": 0,
        "2_0_to_2_9": 0,
        "3_0_to_3_9": 0,
        "4_0_and_above": 0,
        "empty_magnitude": 0,
        "total": len(events),
    }
    for ev in events:
        m_str = ev["magnitude"]
        if m_str == "":
            bands["empty_magnitude"] += 1
        else:
            m = float(m_str)
            if m < 2.0:
                bands["below_2_0"] += 1
            elif m < 3.0:
                bands["2_0_to_2_9"] += 1
            elif m < 4.0:
                bands["3_0_to_3_9"] += 1
            else:
                bands["4_0_and_above"] += 1
    return bands


def top_ten_text(events: list) -> list:
    with_mag = []
    for ev in events:
        if ev["magnitude"] != "":
            with_mag.append(
                (
                    -float(ev["magnitude"]),
                    parse_iso_utc_to_epoch_ms(ev["time_utc"]),
                    ev["event_id"],
                    ev,
                )
            )
    with_mag.sort(key=lambda x: (x[0], x[1], x[2]))
    return [item[3] for item in with_mag[:10]]


def top_ten_geojson(geo_events: list) -> tuple:
    # Raw top 10 as returned by orderby=magnitude (excluding null mag if any)
    non_empty = [e for e in geo_events if e["mag"] is not None]
    raw_top10 = non_empty[:10]
    # Tie-broken top 10 using identical rule: (-mag, earlier time_ms, event_id)
    tie_sorted = sorted(non_empty, key=lambda e: (-e["mag"], e["time_ms"], e["event_id"]))
    tie_top10 = tie_sorted[:10]
    return tie_top10, raw_top10


def compare_routes(week_start: str, txt_top10: list, geo_tie_top10: list, geo_raw_top10: list) -> list:
    disagreements = []
    max_len = max(len(txt_top10), len(geo_tie_top10))
    for i in range(max_len):
        t = txt_top10[i] if i < len(txt_top10) else None
        g = geo_tie_top10[i] if i < len(geo_tie_top10) else None
        gr = geo_raw_top10[i] if i < len(geo_raw_top10) else None
        if t is None or g is None:
            disagreements.append(
                {
                    "week_start": week_start,
                    "rank": i + 1,
                    "type": "missing_rank",
                    "text_id": t["event_id"] if t else "",
                    "geojson_id": g["event_id"] if g else "",
                    "detail": "Different number of top events (<10)",
                }
            )
            continue
        id_diff = t["event_id"] != g["event_id"]
        t_mag = float(t["magnitude"]) if t["magnitude"] != "" else None
        g_mag = g["mag"]
        mag_diff = (t_mag != g_mag) and (abs((t_mag or 0.0) - (g_mag or 0.0)) > 1e-9)
        t_dep = float(t["depth_km"]) if t["depth_km"] != "" else None
        g_dep = g["depth"]
        # Check both exact float equality and 1e-6 rounding difference
        dep_exact_diff = t_dep != g_dep
        if id_diff or mag_diff or dep_exact_diff:
            disagreements.append(
                {
                    "week_start": week_start,
                    "rank": i + 1,
                    "type": "tie_broken_comparison",
                    "text_id": t["event_id"],
                    "geojson_id": g["event_id"],
                    "text_mag": t["magnitude"],
                    "geojson_mag": g["mag_raw"],
                    "text_depth": t["depth_km"],
                    "geojson_depth": g["depth_raw"],
                    "detail": (
                        f"id_diff={id_diff}, mag_diff={mag_diff}, "
                        f"depth_diff={dep_exact_diff} (text={t['depth_km']} vs geojson={g['depth_raw']})"
                    ),
                }
            )
        # Also note if raw server order differed on a magnitude tie before tie-breaking
        if gr is not None and gr["event_id"] != t["event_id"] and not id_diff:
            disagreements.append(
                {
                    "week_start": week_start,
                    "rank": i + 1,
                    "type": "server_tie_order_only",
                    "text_id": t["event_id"],
                    "geojson_id": gr["event_id"],
                    "text_mag": t["magnitude"],
                    "geojson_mag": gr["mag_raw"],
                    "text_depth": t["depth_km"],
                    "geojson_depth": gr["depth_raw"],
                    "detail": (
                        f"Tied magnitude {t['magnitude']}: USGS raw orderby=magnitude placed {gr['event_id']} "
                        f"at rank {i+1}, whereas earlier-time tie-breaker places {t['event_id']} at rank {i+1}."
                    ),
                }
            )
    return disagreements


def make_excerpt(route_type: str, resp: dict, strongest_id: str = "") -> str:
    body = resp["text"]
    if route_type == "count":
        return body.strip()
    if route_type == "text":
        all_lines = body.splitlines()
        head_slice = "\n".join(all_lines[:6])
        if strongest_id:
            for idx, ln in enumerate(all_lines):
                if ln.startswith(strongest_id + "|"):
                    s = max(1, idx)
                    e = min(len(all_lines), idx + 2)
                    top_slice = "\n".join(all_lines[s:e])
                    if idx >= 6:
                        return head_slice + "\n...\n" + top_slice
                    break
        return head_slice
    if route_type == "geojson":
        raw_lines = body.splitlines()
        return "\n".join(raw_lines[:2]) if raw_lines else body[:1200]
    return "\n".join(body.splitlines()[:10])


def run_pull(
    start_monday: str,
    num_weeks: int,
    minlat: float,
    maxlat: float,
    minlon: float,
    maxlon: float,
    csv_out: str,
) -> dict:
    start_dt = datetime.date.fromisoformat(start_monday)
    comp_dt = start_dt - datetime.timedelta(days=7)

    week_defs = []
    # Index 0 is comparison week (Week 0)
    for w_idx in range(-1, num_weeks):
        w_start = start_dt + datetime.timedelta(days=7 * w_idx)
        w_end_excl = w_start + datetime.timedelta(days=7)
        w_sun = w_start + datetime.timedelta(days=6)
        week_defs.append(
            {
                "week_num": w_idx + 1,  # 0 for comparison week, 1..12 for main weeks
                "week_start": w_start.isoformat(),
                "starttime": f"{w_start.isoformat()}T00:00:00",
                "endtime": f"{w_sun.isoformat()}T23:59:59.999",
                "end_exclusive_iso": f"{w_end_excl.isoformat()}T00:00:00",
                "end_exclusive_ms": parse_iso_utc_to_epoch_ms(f"{w_end_excl.isoformat()}T00:00:00Z"),
            }
        )

    requests_log = []
    weekly_results = []
    all_csv_rows = []
    seen_ids = set()
    duplicate_ids = []

    for wd in week_defs:
        base_params = [
            ("starttime", wd["starttime"]),
            ("endtime", wd["endtime"]),
            ("minlatitude", str(minlat)),
            ("maxlatitude", str(maxlat)),
            ("minlongitude", str(minlon)),
            ("maxlongitude", str(maxlon)),
        ]
        txt_params = [("format", "text")] + base_params
        cnt_params = list(base_params)

        r_txt = fetch_usgs(QUERY_PATH, txt_params)
        if not r_txt["ok"]:
            raise RuntimeError(f"USGS text route failed for {wd['week_start']}: {r_txt.get('error')}")
        txt_events = parse_text_events(r_txt["text"], wd["week_start"], wd["end_exclusive_iso"])
        t_top10_preview = top_ten_text(txt_events)
        strongest_id = t_top10_preview[0]["event_id"] if t_top10_preview else ""
        requests_log.append(
            {
                "week_num": wd["week_num"],
                "week_start": wd["week_start"],
                "route": "text",
                "url": r_txt["url"],
                "retrieved_utc": r_txt["retrieved_utc"],
                "http_status": r_txt["http_status"],
                "http_date": r_txt["http_date"],
                "ok": r_txt["ok"],
                "raw_line_count": len(r_txt["text"].splitlines()),
                "parsed_event_count": len(txt_events),
                "excerpt": make_excerpt("text", r_txt, strongest_id),
            }
        )

        r_cnt = fetch_usgs(COUNT_PATH, cnt_params)
        requests_log.append(
            {
                "week_num": wd["week_num"],
                "week_start": wd["week_start"],
                "route": "count",
                "url": r_cnt["url"],
                "retrieved_utc": r_cnt["retrieved_utc"],
                "http_status": r_cnt["http_status"],
                "http_date": r_cnt["http_date"],
                "ok": r_cnt["ok"],
                "excerpt": make_excerpt("count", r_cnt),
            }
        )
        if not r_cnt["ok"]:
            raise RuntimeError(f"USGS count route failed for {wd['week_start']}: {r_cnt.get('error')}")

        cnt_val = int(r_cnt["text"].strip())
        bands = compute_bands(txt_events)

        if wd["week_num"] == 0:
            weekly_results.append(
                {
                    "week_num": 0,
                    "week_start": wd["week_start"],
                    "starttime": wd["starttime"],
                    "endtime": wd["endtime"],
                    "end_exclusive_iso": wd["end_exclusive_iso"],
                    "text_count": len(txt_events),
                    "count_endpoint": cnt_val,
                    "bands": bands,
                }
            )
            continue

        # Main 12 weeks: also pull format=geojson&orderby=magnitude
        geo_params = [("format", "geojson"), ("orderby", "magnitude")] + base_params
        r_geo = fetch_usgs(QUERY_PATH, geo_params)
        requests_log.append(
            {
                "week_num": wd["week_num"],
                "week_start": wd["week_start"],
                "route": "geojson",
                "url": r_geo["url"],
                "retrieved_utc": r_geo["retrieved_utc"],
                "http_status": r_geo["http_status"],
                "http_date": r_geo["http_date"],
                "ok": r_geo["ok"],
                "excerpt": make_excerpt("geojson", r_geo),
            }
        )
        if not r_geo["ok"]:
            raise RuntimeError(f"USGS geojson route failed for {wd['week_start']}: {r_geo.get('error')}")

        geo_meta_cnt, geo_gen_ms, geo_events = parse_geojson_events(r_geo["text"], wd["end_exclusive_ms"])
        t_top10 = top_ten_text(txt_events)
        g_tie_top10, g_raw_top10 = top_ten_geojson(geo_events)
        disagreements = compare_routes(wd["week_start"], t_top10, g_tie_top10, g_raw_top10)

        prev_bands = weekly_results[-1]["bands"]
        prev_total = prev_bands["total"]
        cur_total = bands["total"]
        total_diff = cur_total - prev_total
        pct_change = ((total_diff / prev_total) * 100.0) if prev_total > 0 else None

        wow = {
            "prev_week_start": weekly_results[-1]["week_start"],
            "prev_total": prev_total,
            "below_2_0_diff": bands["below_2_0"] - prev_bands["below_2_0"],
            "2_0_to_2_9_diff": bands["2_0_to_2_9"] - prev_bands["2_0_to_2_9"],
            "3_0_to_3_9_diff": bands["3_0_to_3_9"] - prev_bands["3_0_to_3_9"],
            "4_0_and_above_diff": bands["4_0_and_above"] - prev_bands["4_0_and_above"],
            "empty_magnitude_diff": bands["empty_magnitude"] - prev_bands["empty_magnitude"],
            "total_diff": total_diff,
            "pct_change": round(pct_change, 2) if pct_change is not None else None,
        }

        for ev in txt_events:
            if ev["event_id"] in seen_ids:
                duplicate_ids.append(ev["event_id"])
            seen_ids.add(ev["event_id"])
            all_csv_rows.append(ev)

        weekly_results.append(
            {
                "week_num": wd["week_num"],
                "week_start": wd["week_start"],
                "starttime": wd["starttime"],
                "endtime": wd["endtime"],
                "end_exclusive_iso": wd["end_exclusive_iso"],
                "text_count": len(txt_events),
                "count_endpoint": cnt_val,
                "geojson_count": len(geo_events),
                "geojson_metadata_count": geo_meta_cnt,
                "geojson_generated_ms": geo_gen_ms,
                "bands": bands,
                "wow": wow,
                "top10_text": t_top10,
                "top10_geojson_tie": g_tie_top10,
                "top10_geojson_raw": g_raw_top10,
                "disagreements": disagreements,
            }
        )

    os.makedirs(os.path.dirname(os.path.abspath(csv_out)), exist_ok=True)
    with open(csv_out, "w", encoding="utf-8", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=CSV_COLUMNS, lineterminator="\n")
        writer.writeheader()
        for row in all_csv_rows:
            writer.writerow(row)

    csv_bytes = open(csv_out, "rb").read()
    csv_sha256 = hashlib.sha256(csv_bytes).hexdigest()

    return {
        "pull_completed_utc": datetime.datetime.now(datetime.timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ"),
        "first_http_date": requests_log[0]["http_date"] if requests_log else "",
        "last_http_date": requests_log[-1]["http_date"] if requests_log else "",
        "csv_path": os.path.abspath(csv_out),
        "csv_bytes": len(csv_bytes),
        "csv_sha256": csv_sha256,
        "csv_row_count": len(all_csv_rows),
        "duplicate_ids": duplicate_ids,
        "comp_start": comp_dt.isoformat(),
        "weekly_results": weekly_results,
        "requests_log": requests_log,
    }


def compute_csv_changes(pull1_csv_path: str, pull2_csv_path: str) -> dict:
    def load_map(path):
        rows = {}
        with open(path, "r", encoding="utf-8", newline="") as f:
            reader = csv.DictReader(f)
            for r in reader:
                rows[r["event_id"]] = r
        return rows

    m1 = load_map(pull1_csv_path)
    m2 = load_map(pull2_csv_path)
    k1 = set(m1.keys())
    k2 = set(m2.keys())
    new_keys = sorted(k2 - k1)
    gone_keys = sorted(k1 - k2)
    common_keys = sorted(k1 & k2)

    changed_items = []
    for k in common_keys:
        r1, r2 = m1[k], m2[k]
        diffs = {}
        for col in CSV_COLUMNS[1:]:
            if r1[col] != r2[col]:
                diffs[col] = {"pull1": r1[col], "pull2": r2[col]}
        if diffs:
            changed_items.append({"event_id": k, "week_start": r2["week_start"], "diffs": diffs})

    assert len(m2) == len(m1) + len(new_keys) - len(gone_keys), "Row count identity mismatch!"
    return {
        "pull1_rows": len(m1),
        "pull2_rows": len(m2),
        "new_count": len(new_keys),
        "gone_count": len(gone_keys),
        "changed_count": len(changed_items),
        "new_keys": new_keys[:50],
        "gone_keys": gone_keys[:50],
        "changed_items": changed_items[:100],
    }
